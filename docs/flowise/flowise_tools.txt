
# 1. Place Order Tool (Menu Agent)
Name: place_order
Description: Place a new order for items. IMPORTANT: Ask the user for their Table Number first.
Schema:
{
    "type": "object",
    "properties": {
        "items": { "type": "array", "items": { "type": "object", "properties": { "menu_item_id": { "type": "string" }, "quantity": { "type": "number" }, "notes": { "type": "string" } } } },
        "table_number": { "type": "string" }
    },
    "required": ["items", "table_number"]
}

# 4. Verify Payment Tool (Menu Agent)
Name: verify_payment
Description: Verify a bank transaction and mark the order as paid.
Schema:
{
    "type": "object",
    "properties": {
        "transaction_id": { "type": "string", "description": "Transaction Reference Number" },
        "bank": { "type": "string", "enum": ["telebirr", "cbe", "dashen", "abyssinia"], "description": "Bank Name" },
        "order_id": { "type": "string", "description": "The Order UUID to update (Optional but recommended)" }
    },
    "required": ["transaction_id", "bank"]
}

# 6. Manage Menu Tool (Master Agent)
Name: manage_menu
Description: Add, Update, or Delete Menu Items. Restricted to Managers.
Schema:
{
    "type": "object",
    "properties": {
        "action": { "type": "string", "enum": ["update", "delete"], "description": "Action to perform (use 'update' for adding new items or changing existing ones)" },
        "item": { "type": "object", "description": "Item details (name, price, category) for updates" },
        "target_id": { "type": "string", "description": "UUID of item to delete" },
        "user_id": { "type": "string", "description": "User ID of the manager performing the action" }
    },
    "required": ["action", "user_id"]
}
